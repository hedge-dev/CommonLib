333///
334/// Installs a hook defined with \ref HOOK or \ref ASM_HOOK at an explicit address.
335///
336/// \param FUNCTION_NAME The name of the function to call before the original.
337/// \param ADDRESS The address of the function to hook.

344 if (!original_##FUNCTION_NAME && !(ADDRESS))
345 return false;
346
347 *(void**)&original_##FUNCTION_NAME = (void*)(ADDRESS);
348
349 DetourTransactionBegin();
350 DetourUpdateThread(GetCurrentThread());
351 DetourAttach((void**)&original_##FUNCTION_NAME, &impl_##FUNCTION_NAME);

424 inline void* GetPostHookAddress(void* in_pHookStart)
425 {
426 if (!in_pHookStart)
427 return nullptr;
428
429 const auto branchInfo = hedgedev::csl::mem::GetBranchInfo(in_pHookStart);
430
431 auto pPostHook = (uint8_t*)(size_t(in_pHookStart) + branchInfo.InstrLength);
432
433 while (*pPostHook == 0xCC)
434 pPostHook++;
435
436 return pPostHook;
