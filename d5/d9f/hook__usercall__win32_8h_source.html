462 ///
463 /// Emits a trampoline that forwards arguments from a function with custom calling
464 /// convention to a `__cdecl` function in a hook defined with \ref USER_HOOK.
465 ///
466 /// \param in_info The info about the hook.
467 /// \param in_address The address of the `__cdecl` function to trampoline to.
468 /// \param in_to_original Determines whether this trampoline is going back to the
469 /// original function.
470 ///
471 /// \returns A pointer to the trampoline that was emitted.
472 ///
473 inline void* emit_user_trampoline(const usercall_info& in_info, void* in_address, bool in_to_original = false)
474 {
475 std::vector<uint8_t> result{};
476
477 const auto emit_imm8 = [&](int in_value) -> size_t
478 {
479 auto pos = result.size();
480
481 result.push_back(in_value & 0xFF);
482
483 return pos;
484 };
485
486 const auto emit_imm16 = [&](int in_value) -> size_t
487 {
488 auto pos = emit_imm8(in_value);
489
490 result.push_back((in_value >> 8) & 0xFF);
491
492 return pos;
493 };
494
495 const auto emit_imm32 = [&](int in_value) -> size_t
496 {
497 auto pos = emit_imm16(in_value);
498
499 result.push_back((in_value >> 16) & 0xFF);
500 result.push_back((in_value >> 24) & 0xFF);
501
502 return pos;
503 };
504
505 const auto emit_push = [&](user_register in_register, std::optional<int> in_offset = {})
506 {
507 const auto register_id = get_register_id(in_register);
508
509 if (in_offset.has_value())
510 {
511 const auto offset = in_offset.value();
512 const auto emit_ebp_no_offset = in_register == user_register::EBP && offset == 0;
513
514 result.push_back(0xFF);
515
516 if (!offset && !emit_ebp_no_offset)
517 {
518 // push [{register}]
519 result.push_back(uint8_t(0x30 | register_id));
520 }
521 else if (offset <= 0x7F || emit_ebp_no_offset)
522 {
523 // push [{register} + {imm8}]
524 result.push_back(uint8_t(0x70 | register_id));
525 }
526 else
527 {
528 // push [{register} + {imm32}]
529 result.push_back(uint8_t(0xB0 | register_id));
530 }
531
532 if (in_register == user_register::ESP)
533 result.push_back(0x24);
534
535 if (!offset && !emit_ebp_no_offset)
536 return;
537
538 if ((offset > 0 && offset <= 0x7F) || emit_ebp_no_offset)
539 {
540 emit_imm8(offset);
541 }
542 else if (offset > 0x7F)
543 {
544 emit_imm32(offset);
545 }
546
547 return;
548 }
549
550 // push {register}
551 result.push_back(uint8_t(0x50 | register_id));
552 };
553
554 const auto emit_pop = [&](user_register in_register)
555 {
556 // pop {register}
557 result.push_back(uint8_t(0x58 | get_register_id(in_register)));
558 };

560 const auto emit_add_subtract = [&](user_register in_register, int in_value)
561 {
562 if (!in_value)
563 return;
564
565 const auto register_id = get_register_id(in_register);
566 const auto is_add = in_value > 0;
567 const auto value_abs = std::abs(in_value);
568
569 if (value_abs == 1)
570 {
571 auto opcode = 0x40;
572
573 if (!is_add)
574 opcode |= 8;
575
576 // inc/dec {register}
577 result.push_back(uint8_t(opcode | register_id));
578 }
579 else
580 {
581 auto modifier = 0;
582
583 if (!is_add)
584 modifier = 0x28;
585
586 if (value_abs > 0 && value_abs <= 0x7F)
587 {
588 // add/sub {register}, {imm8}
589 result.push_back(0x83);
590 result.push_back(uint8_t(0xC0 | modifier | register_id));
591 emit_imm8(value_abs);
592 }
593 else if (value_abs > 0x7F)
594 {
595 // add/sub {register}, {imm32}
596 if (in_register == user_register::EAX)
597 {
598 result.push_back(uint8_t(0x05 | modifier));
599 }
600 else
601 {
602 result.push_back(0x81);
603 result.push_back(uint8_t(0xC0 | modifier | register_id));
604 }
605
606 emit_imm32(value_abs);
607 }
608 }

611 const auto emit_load_store_sse = [&](user_register in_dst, user_register in_src, int in_offset = 0, int in_float_size = sizeof(float), bool in_store = false)
612 {
613 const auto dst_register_id = get_register_id(in_dst);
614 const auto src_register_id = get_register_id(in_src);
615
616 const auto ptr_register = in_store ? in_dst : in_src;
617 const auto ptr_register_id = in_store ? dst_register_id : src_register_id;
618
619 const auto data_register = in_store ? in_src : in_dst;
620 const auto data_register_id = in_store ? src_register_id : dst_register_id;
621
622 const auto emit_ebp_no_offset = ptr_register == user_register::EBP && in_offset == 0;
623
624 // movss dword ptr
625 auto opcode = 0xF3;
626
627 // movsd qword ptr
628 if (in_float_size == sizeof(double))
629 opcode = 0xF2;
630
631 result.push_back(opcode);
632 result.push_back(0x0F);
633 result.push_back(in_store ? 0x11 : 0x10);
634
635 auto modifier = 0;
636
637 if ((in_offset > 0 && in_offset <= 0x7F) || emit_ebp_no_offset)
638 {
639 modifier = 0x40;
640 }
641 else if (in_offset > 0x7F)
642 {
643 modifier = 0x80;
644 }
645
646 if (ptr_register == user_register::ESP)
647 {
648 result.push_back(uint8_t(0x04 | modifier | (data_register_id << 3)));
649 result.push_back(0x24);
650 }
651 else
652 {
653 result.push_back(uint8_t(modifier | (ptr_register_id << 3) | data_register_id));
654 }
655
656 if ((in_offset > 0 && in_offset <= 0x7F) || emit_ebp_no_offset)
657 {
658 // movss/movsd {dst}, dword/qword ptr [{src} + {imm8}]
659 // movss/movsd dword/qword ptr [{dst} + {imm8}], {src}
660 emit_imm8(in_offset);
661 }
662 else if (in_offset > 0x7F)
663 {
664 // movss/movsd {dst}, dword/qword ptr [{src} + {imm32}]
665 // movss/movsd dword/qword ptr [{dst} + {imm32}], {src}

744 const auto emit_load_store = [&](user_register in_dst, user_register in_src, int in_offset = 0, bool in_store = false)
745 {

747 const auto src_register_id = get_register_id(in_src);
748
749 const auto ptr_register = in_store ? in_dst : in_src;
750 const auto ptr_register_id = in_store ? dst_register_id : src_register_id;

752 const auto data_register = in_store ? in_src : in_dst;
753 const auto data_register_id = in_store ? src_register_id : dst_register_id;

759
760 // mov {dst}, [{src}]

762 opcode |= 2;

771 }

774 modifier = 0x80;
775 }

783 {
784 result.push_back(uint8_t(modifier | (ptr_register_id << 3) | data_register_id));
785 }
786
787 if ((in_offset > 0 && in_offset <= 0x7F) || emit_ebp_no_offset)
788 {
789 // mov {dst}, [{src} + {imm8}]
790 // mov [{dst} + {imm8}], {src}
791 emit_imm8(in_offset);
792 }
793 else if (in_offset > 0x7F)
794 {
795 // mov {dst}, [{src} + {imm32}]
796 // mov [{dst} + {imm32}], {src}
797 emit_imm32(in_offset);
798 }

801 const auto emit_move_register_fpu = [&](user_register in_dst, user_register in_src, int in_float_size = sizeof(float))
802 {
803 const auto dst_register_family = get_register_family(in_dst);
804 const auto src_register_family = get_register_family(in_src);
805
806 if (dst_register_family == user_register_family::FPU)
807 {
808 if (dst_register_family == src_register_family)
809 {
810 if (in_dst == in_src)
811 return;
812
813 // Exchange ST(0) with source register.
814 emit_exchange_register_fpu(in_src);
815
816 // Copy ST(0) to destination register.
817 emit_load_store_fpu(in_dst);
818
819 // Restore ST(0).
820 emit_exchange_register_fpu(in_src);
821 }
822 else
823 {
824 // Allocate stack space for register.
825 emit_add_subtract(user_register::ESP, -in_float_size);
826
827 if (src_register_family == user_register_family::GPR)
828 {
829 // Copy GPR register into stack.
830 emit_load_store(user_register::ESP, in_src, 0, true);
831 }
832 else if (src_register_family == user_register_family::SSE)
833 {
834 // Copy SSE register into stack.
835 emit_load_store_sse(user_register::ESP, in_src, 0, in_float_size, true);
836 }
837
838 // Exchange ST(0) with destination register.
839 if (in_dst != user_register::ST0)
840 emit_exchange_register_fpu(in_dst);
841
842 // Load stack value into FPU register.
843 emit_load_store_fpu(user_register::ESP, 0, in_float_size);
844
845 // Restore ST(0).
846 if (in_dst != user_register::ST0)
847 emit_exchange_register_fpu(in_dst);
848
849 // Deallocate stack space.
850 emit_add_subtract(user_register::ESP, in_float_size);
851 }
852 }
853 else
854 {
855 // Allocate stack space for FPU register.
856 emit_add_subtract(user_register::ESP, -in_float_size);
857
858 // Exchange ST(0) with source register.
859 if (in_src != user_register::ST0)
860 emit_exchange_register_fpu(in_src);
861
862 // Copy ST(0) into stack.
863 emit_load_store_fpu(user_register::ESP, 0, in_float_size, true);
864
865 // Restore ST(0).
866 if (in_src != user_register::ST0)
867 emit_exchange_register_fpu(in_src);
868
869 if (dst_register_family == user_register_family::GPR)
870 {
871 // Load stack value into GPR register.
872 emit_load_store(in_dst, user_register::ESP, 0);
873 }
874 else if (dst_register_family == user_register_family::SSE)
875 {
876 // Load stack value into SSE register.
877 emit_load_store_sse(in_dst, user_register::ESP, 0, in_float_size);
878 }
879
880 // Deallocate stack space.
881 emit_add_subtract(user_register::ESP, in_float_size);
882 }
